    password VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

**Tasks Table**
```sql
CREATE TABLE tasks (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    title VARCHAR(100) NOT NULL,
    description TEXT,
    status VARCHAR(20) CHECK (status IN ('pending', 'in_progress', 'completed')) DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

### 5. Key Design Decisions

- **Authentication**: Using JWT for stateless authentication, allowing easy scalability and horizontal scaling.
- **Database**: Choosing a relational database for structured data and relationships between users and tasks.
- **Error Handling**: Centralized error handling middleware to manage API responses uniformly.
- **Validation**: Using a validation library (like Joi or express-validator) to ensure data integrity.

### 6. Edge Cases

- **User Registration**: Handle cases where the username already exists.
- **Task Operations**: Ensure that tasks cannot be updated or deleted by unauthorized users.
- **Empty Requests**: Validate that required fields are provided in requests.
- **Invalid Task Status**: Ensure that only valid statuses can be assigned to tasks.
- **Database Connection Issues**: Gracefully handle database connection failures.

### 7. Scalability Considerations

- **Load Balancing**: Use load balancers to distribute incoming traffic across multiple instances of the backend service.
- **Database Optimization**: Implement indexing on frequently queried fields (e.g., user_id in tasks).
- **Caching**: Use caching mechanisms (like Redis) for frequently accessed data to reduce database load.
- **Microservices**: Consider breaking down the application into microservices if the system grows significantly.
- **Horizontal Scaling**: Design the application to support horizontal scaling by using stateless services.

This architecture and design outline provides a solid foundation for building a production-ready task management system.

### 2. Services

**services/authService.js**
```javascript
const User = require('../models/User');
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');

class AuthService {
    async register(username, password) {
        const hashedPassword = await bcrypt.hash(password, 10);
        return User.create({ username, password: hashedPassword });
    }

    async login(username, password) {
        const user = await User.findOne({ where: { username } });
        if (!user || !(await bcrypt.compare(password, user.password))) {
            throw new Error('Invalid credentials');
        }
        const token = jwt.sign({ id: user.id }, process.env.JWT_SECRET, { expiresIn: '1h' });
        return { token };
    }
}

module.exports = new AuthService();
```

**services/taskService.js**
```javascript
const Task = require('../models/Task');

class TaskService {
    async getAllTasks(userId) {
        return Task.findAll({ where: { user_id: userId } });
    }

    async createTask(userId, title, description) {
        return Task.create({ user_id: userId, title, description });
    }

    async updateTask(id, updates) {
        const task = await Task.findByPk(id);
        if (!task) throw new Error('Task not found');
        return task.update(updates);
    }

    async deleteTask(id) {
        const task = await Task.findByPk(id);
        if (!task) throw new Error('Task not found');
        return task.destroy();
    }
}

module.exports = new TaskService();
```

### 5. Middleware

**middleware/authMiddleware.js**
```javascript
const jwt = require('jsonwebtoken');

const authenticate = (req, res, next) => {
    const token = req.headers['authorization']?.split(' ')[1];
    if (!token) return res.status(401).json({ error: 'No token provided' });

    jwt.verify(token, process.env.JWT_SECRET, (err, decoded) => {
        if (err) return res.status(401).json({ error: 'Unauthorized' });
        req.user = decoded; // Attach user info to request
        next();
    });
};

module.exports = authenticate;
```
